class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self, slow_query_threshold: float = 1.0):
        self.slow_query_threshold = slow_query_threshold
        self.stats = defaultdict(QueryStats)
        self._lock = threading.Lock()
        
    def record_query(self, query_type: str, duration: float, success: bool):
        """记录查询性能数据"""
        with self._lock:
            self.stats[query_type].query_count += 1
            self.stats[query_type].total_time += duration
            
            if not success:
                self.stats[query_type].error_count += 1
                
            if duration > self.slow_query_threshold:
                self.stats[query_type].slow_queries.append({
                    'duration': duration,
                    'timestamp': time.time(),
                    'type': query_type
                })
                
    def get_stats(self) -> Dict[str, Dict]:
        """获取性能统计信息"""
        with self._lock:
            result = {}
            for query_type, stats in self.stats.items():
                avg_time = stats.total_time / stats.query_count if stats.query_count > 0 else 0
                result[query_type] = {
                    'query_count': stats.query_count,
                    'total_time': round(stats.total_time, 3),
                    'avg_time': round(avg_time, 3),
                    'error_count': stats.error_count,
                    'slow_queries': stats.slow_queries[-10:]  # 保留最近10个慢查询
                }
            return result
    
    def reset_stats(self):
        """重置统计数据"""
        with self._lock:
            self.stats.clear()
